import numpy as np
from scipy.io import wavfile
import os

asset_path = '/var/www/vegas-saas/assets/audio/'
if not os.path.exists(asset_path):
    os.makedirs(asset_path)

def generate_revenue_ping(fs=44100):
    t = np.linspace(0, 1.5, int(fs * 1.5))
    wave = np.sin(2 * np.pi * 987.77 * t) * np.exp(-3 * t)
    wave += 0.5 * np.sin(2 * np.pi * 1975.54 * t) * np.exp(-6 * t)
    return (wave * 32767).astype(np.int16)

def generate_system_hardened(fs=44100):
    t = np.linspace(0, 2.0, int(fs * 2.0))
    wave = np.sign(np.sin(2 * np.pi * 110 * t)) * np.exp(-2 * t)
    wave *= (1 - 0.5 * t)
    return (wave * 15000).astype(np.int16)

def generate_flux_complete(fs=44100):
    t = np.linspace(0, 2.0, int(fs * 2.0))
    freq = np.linspace(440, 1760, len(t))
    wave = np.sin(2 * np.pi * freq * t) * np.exp(-1.5 * t)
    return (wave * 25000).astype(np.int16)

fs = 44100
wavfile.write(os.path.join(asset_path, 'revenue_ping.wav'), fs, generate_revenue_ping(fs))
wavfile.write(os.path.join(asset_path, 'system_hardened.wav'), fs, generate_system_hardened(fs))
wavfile.write(os.path.join(asset_path, 'flux_complete.wav'), fs, generate_flux_complete(fs))
print(f"Assets successfully minted in {asset_path}")
