const PORT = 3021;
const OLLAMA_URL = "http://localhost:11434";
const PROMPTS = JSON.parse(fs.readFileSync(path.join(__dirname, "prompts.json"), "utf-8"));

async function callOllama(model, messages) {
  return new Promise((resolve, reject) => {
    const payload = JSON.stringify({ model, messages, stream: false, format: "json" });
    const opts = {
      hostname: "localhost", port: 11434, path: "/api/chat",
      method: "POST", headers: { "Content-Type": "application/json" }
    };
    const req = http.request(opts, (res) => {
      let data = "";
      res.on("data", (chunk) => data += chunk);
      res.on("end", () => {
        try { resolve(JSON.parse(data)); } catch (e) { reject(new Error(`Parse error: ${data.slice(0, 200)}`)); }
      });
    });
    req.on("error", reject);
    req.write(payload);
    req.end();
  });
}

function buildProductLaunchMessages(productName, pricePoint, coreBenefit) {
  const userPrompt = PROMPTS.prompts.product_launch.template
    .replace("{{product_name}}", productName)
    .replace("{{price_point}}", pricePoint)
    .replace("{{core_benefit}}", coreBenefit);
  return [
    { role: "system", content: PROMPTS.prompts.system.content },
    { role: "system", content: PROMPTS.prompts.cromenix_specialization.content },
    { role: "user", content: userPrompt }
  ];
}

const server = http.createServer(async (req, res) => {
  res.setHeader("Content-Type", "application/json");
  res.setHeader("Access-Control-Allow-Origin", "*");
  res.setHeader("Access-Control-Allow-Methods", "GET, POST, OPTIONS");
  res.setHeader("Access-Control-Allow-Headers", "Content-Type");

  if (req.method === "OPTIONS") { res.writeHead(204); return res.end(); }

  if (req.url === "/health" && req.method === "GET") {
    return res.end(JSON.stringify({
      status: "operational", service: "content-spark",
      territory: PROMPTS.territory, brand: PROMPTS.brand,
      ollama: OLLAMA_URL, timestamp: new Date().toISOString()
    }));
  }

  if (req.url === "/api/generate/product-launch" && req.method === "POST") {
    let body = "";
    req.on("data", (chunk) => body += chunk);
    req.on("end", async () => {
      try {
        const { product_name, price_point, core_benefit, model } = JSON.parse(body);
        if (!product_name || !price_point || !core_benefit) {
          res.writeHead(400);
          return res.end(JSON.stringify({ error: "Missing required fields: product_name, price_point, core_benefit" }));
        }
        const messages = buildProductLaunchMessages(product_name, price_point, core_benefit);
        const result = await callOllama(model || "llama3", messages);
        const content = result.message?.content || "{}";
        let parsed;
        try { parsed = JSON.parse(content); } catch { parsed = { raw: content }; }
        res.writeHead(200);
        return res.end(JSON.stringify({
          territory: PROMPTS.territory, brand: PROMPTS.brand,
          input: { product_name, price_point, core_benefit },
          output: parsed, model: model || "llama3",
          revenue_split: { operator: 70, core: 30 },
          timestamp: new Date().toISOString()
        }));
      } catch (err) {
        res.writeHead(500);
        return res.end(JSON.stringify({ error: err.message }));
      }
    });
    return;
  }

  if (req.url === "/api/generate/custom" && req.method === "POST") {
    let body = "";
    req.on("data", (chunk) => body += chunk);
    req.on("end", async () => {
      try {
        const { prompt, model } = JSON.parse(body);
        const messages = [
          { role: "system", content: PROMPTS.prompts.system.content },
          { role: "system", content: PROMPTS.prompts.cromenix_specialization.content },
          { role: "user", content: prompt }
        ];
        const result = await callOllama(model || "llama3", messages);
        res.writeHead(200);
        return res.end(JSON.stringify({
          territory: PROMPTS.territory, output: result.message?.content,
          model: model || "llama3", timestamp: new Date().toISOString()
        }));
      } catch (err) {
        res.writeHead(500);
        return res.end(JSON.stringify({ error: err.message }));
      }
    });
    return;
  }

  res.writeHead(404);
  res.end(JSON.stringify({ error: "Not found", routes: ["/health", "/api/generate/product-launch", "/api/generate/custom"] }));
});

server.listen(PORT, () => {
  console.log(`[Content-Spark] Cromenix Sovereign AI running on port ${PORT}`);
  console.log(`[Content-Spark] Territory: ${PROMPTS.territory}`);
  console.log(`[Content-Spark] Brand: ${PROMPTS.brand.primary} / ${PROMPTS.brand.secondary}`);
  console.log(`[Content-Spark] Ollama: ${OLLAMA_URL}`);
});
